import { BaseEvent, EVENT_TYPES } from '../base-event';
 
/**
 * Payload for the ConversationCreated event.
 *
 * Published by: Conversation Engine when a brand-new Conversation is persisted.
 * Consumed by: Sales Engine (to initialize lead status), Analytics Engine.
 */
export interface ConversationCreatedPayload {
  /** Internal UUID of the newly created Conversation. */
  readonly conversationId: string;
 
  /** Internal UUID of the Contact who initiated the conversation. */
  readonly contactId: string;
 
  /** The channel through which the conversation was initiated. */
  readonly channelId: string;
 
  /** ISO-8601 creation timestamp from the database record. */
  readonly createdAt: string;
}
 
export class ConversationCreatedEvent extends BaseEvent<ConversationCreatedPayload> {
  constructor(tenantId: string, payload: ConversationCreatedPayload) {
    super(EVENT_TYPES.CONVERSATION_CREATED, tenantId, payload);
  }
}
